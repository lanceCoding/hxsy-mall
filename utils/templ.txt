写一个小程序端根据省份和数量（item）计算运费的函数


// 在小程序页面中获取运费模板数据
const db = wx.cloud.database();
let shippingTemplate = null;

async function loadShippingTemplate() {
  try {
    const res = await db.collection('shipping_template')
      .where({ enabled: true })
      .orderBy('priority', 'asc')
      .limit(1)
      .get();
    
    if (res.data.length > 0) {
      shippingTemplate = res.data[0];
    }
  } catch (err) {
    console.error('获取运费模板失败:', err);
  }
}
